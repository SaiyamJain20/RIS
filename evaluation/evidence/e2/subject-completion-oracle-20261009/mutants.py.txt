import copy,hashlib,json,pathlib
from evaluation.tests.test_subject_completion import facts
from evaluation.oracles.subject import check_subject_completion
source=pathlib.Path('evaluation/oracles/subject.py').read_text()
controls=[('native-preservation', 'require(after_release == expected_after,', 'require(True,'),
          ('bob-completion', "require(bob_completion['ok'] is True and bob_completion['result'] == expected_bob,", 'require(True,'),
          ('alice-refusal', "require(alice_completion['ok'] is False and alice_completion['error']['code'] == 'WRONG_SUBJECT',", 'require(True,')]
rows=[]
for label,old,new in controls:
 assert source.count(old)==1
 for profile in ['paper-strict','paper-minimal']:
  d=facts(profile)
  if label=='native-preservation':
   d['after_release']['logs']['pbc'].append({'name':'OutcomeRecorded'})
   d['after_completion']=copy.deepcopy(d['after_release'])
  elif label=='bob-completion': d['bob_completion']['result']['decryptable']=False
  else: d['alice_completion']['error']['code']='EXPIRED'
  rejected=False
  try: check_subject_completion(**d)
  except ValueError: rejected=True
  assert rejected
  namespace={}; exec(compile(source.replace(old,new),'deliberate-in-memory-mutant','exec'),namespace)
  accepted=namespace['check_subject_completion'](**d)['ok'];assert accepted
  rows.append(dict(control=label,profile=profile,realOracleRejected=True,deliberateMutantAccepted=True))
p=pathlib.Path(__file__).parent/'mutants-result.json'
p.write_text(json.dumps(dict(sourceSha256=hashlib.sha256(source.encode()).hexdigest(),rows=rows,scope='deliberate in-memory checker mutations; no owner/runtime change or HMBC finding'),indent=2)+'\n')
print(json.dumps(rows))
