"""Offline reproduction of the report correction; no production code is changed.

Run with the root venv's python -I, giving an exact protocol snapshot and output dir.
The fake ledger view is owner-authored. This is not a native ledger/F case test.
"""
import hashlib
import importlib.metadata
import json
import os
from pathlib import Path
import sqlite3
import sys
import tempfile

protocol, output = map(lambda p: Path(p).resolve(), sys.argv[1:])
output.mkdir(parents=True, exist_ok=True)
sys.dont_write_bytecode = True
sys.path[:0] = [str(protocol), str(protocol / "tests")]
os.environ["PYTEST_DISABLE_PLUGIN_AUTOLOAD"] = "1"
blocked = []


def offline_guard(event, args):
    if event in {"subprocess.Popen", "os.system", "os.posix_spawn", "socket.connect", "socket.bind"}:
        blocked.append(event)
        raise RuntimeError("offline review forbids " + event)


sys.addaudithook(offline_guard)


def manifest():
    return {str(p.relative_to(protocol)): hashlib.sha256(p.read_bytes()).hexdigest()
            for p in sorted(protocol.rglob("*")) if p.is_file()}


before = manifest()
from hmbc_protocol import witness_service as svc, witness_state as ws
import test_witness_service_decision as fixture

probes = []
with tempfile.TemporaryDirectory(prefix="report-review-") as tmp:
    for name, first_result, consumed, advance in [
        ("allow_then_deny_after_grant", True, True, False),
        ("allow_then_deny_same_live_sequence", True, False, False),
        ("allow_then_deny_new_sequence_no_grant", True, False, True),
        ("deny_then_allow_new_sequence_no_grant", False, False, True),
    ]:
        view = fixture.DView()
        view.satisfied = first_result
        store = ws.init_store(str(Path(tmp) / name), fixture.BIND, fixture.B, fixture.OWNER)
        try:
            service = svc.WitnessService(store=store, key=fixture.KEY, view=view,
                                         deployment=fixture.DEPLOY, witness_set_version="2")
            first, _ = fixture.derive(view)
            assert service.sign_decision(first)
            earlier = store.evidence()
            view.satisfied = not first_result
            view.consumed = consumed
            if advance:
                view.cp = view.current = dict(view.cp, sequence="5", sourceBlockNumber="210",
                                             sourceBlockHash=fixture.h("pbc:210"), expiresAt="1045")
            expected = "REPLAY" if consumed else "SIGNED" if advance else "CONFLICT_PINNED"
            try:
                second, _ = fixture.derive(view)
            except svc.WitnessRefusal as exc:
                assert consumed and exc.code == "REPLAY"
                # Exercise the service with a canonical proposal too: it must refuse before the store.
                second = dict(first, result="0")
            try:
                service.sign_decision(second)
                actual = "SIGNED"
            except svc.WitnessRefusal as exc:
                actual = exc.reason or exc.code
            assert actual == expected, (name, actual, expected)
            after = store.evidence()
            assert first["requestDigest"] == second["requestDigest"]
            assert first["result"] != second["result"]
            if advance:
                assert len(after["superseded"]) == 1
                assert after["superseded"][0]["death"]["kind"] == "overtaken-decision-sequence"
            else:
                assert after == earlier
            probes.append({"name": name, "expected": expected, "actual": actual,
                           "firstResult": first["result"], "secondResult": second["result"],
                           "sameRequestDigest": True, "storeUnchanged": after == earlier,
                           "supersededCount": len(after["superseded"])})
        finally:
            store.close()

import pytest
pytest_exit = pytest.main([
    "-c", str(protocol / "pytest.ini"), "--rootdir", str(protocol),
    "-p", "no:cacheprovider", "-q",
    str(protocol / "tests/test_witness_service_decision.py"),
    str(protocol / "tests/test_witness_state.py"),
])
after = manifest()
result = {"classification": "tested reconstruction result; injected ledger view",
          "pytestExit": int(pytest_exit), "probes": probes,
          "sourceBefore": before, "sourceAfter": after, "sourceUnchanged": before == after,
          "blockedOperations": blocked, "servicesStarted": [], "gateF": "NOT_RUN",
          "python": sys.version, "sqlite": sqlite3.sqlite_version,
          "packages": {name: importlib.metadata.version(name)
                       for name in ("pytest", "eth-account", "eth-abi", "jsonschema")}}
(output / "review.json").write_text(json.dumps(result, indent=2) + "\n")
print(json.dumps({k: v for k, v in result.items() if k not in {"sourceBefore", "sourceAfter"}}, indent=2))
assert before == after and not blocked
raise SystemExit(pytest_exit)
