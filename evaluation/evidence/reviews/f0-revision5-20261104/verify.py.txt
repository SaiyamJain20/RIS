def git(*a): return subprocess.check_output(['git',*a],cwd=R)
def dump(name,obj): (O/name).write_text(json.dumps(obj,indent=2)+'\n')

assert git('branch','--show-current').decode().strip()=='agent-3'

    assert hashlib.sha256(git('show',r['commit']+':'+r['path'])).hexdigest()==r['sha256']

assert not d['executionAuthorized'] and not d['actualFResults'] and not d['exactOwnerAcks']

assert all(x['experimentMode']==('ANALYSIS_ONLY' if x['id'] in ('F-25','F-26') else 'INTEGRATED_SYNTHETIC') for x in c.values())
rows=list(csv.DictReader((O/'fault-model.csv').open()))
oldrows=list(csv.DictReader((OLD/'fault-model.csv').open()))
assert len(rows)==14 and len({x['id'] for x in rows})==14
for x,y in zip(rows,oldrows):
    assert all(x[k]==y[k] for k in x if k not in ('source','open_dependency'))
assert all(set(x['components']) <= {r['id'] for r in rows} for x in c.values())
assert (R/d['design']).is_file() and (R/d['faultModel']).is_file()
plan=(O/'SOURCE-INTERFACE-PLAN.md').read_text()

    assert a['owners']['protocol']['blobs'][p] in plan
assert all(name in plan for name in ('A1-0100','A2-0046','H3','H5','H6','F-16','F-17','F-18','F-19','F-20','F-31','F-32','T+31','T+32'))

    assert all(mutant in ' '.join(c[k]['controls']) for k in cases)
assert 'WRONG_DESTINATION/LEGACY_STATE' in c['F-32']['prospectiveSafety']
assert 'before a new-epoch checkpoint' in ' '.join(c['F-28']['controls'])
assert 'after the new-epoch checkpoint' in ' '.join(c['F-28']['controls'])
assert all(s in ' '.join(c['FL-10']['controls']) for s in ('SET_CA_ADMINS','bit 6','bit 7','EXPIRED','REPLAY','MALFORMED'))
