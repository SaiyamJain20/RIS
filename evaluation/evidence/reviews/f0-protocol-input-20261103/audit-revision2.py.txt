import datetime, hashlib, json, pathlib, subprocess, sys, time

R = pathlib.Path('/home/karma/Documents/RIS')
O = R / 'evaluation/evidence/reviews/f0-protocol-input-20261103/revision2'
O.mkdir(parents=True, exist_ok=True)
def git(*args):
    return subprocess.check_output(['git', *args], cwd=R)
def sha(b): return hashlib.sha256(b).hexdigest()
def save(name, obj): (O/name).write_text(json.dumps(obj, indent=2)+'\n')
def now(): return datetime.datetime.now(datetime.timezone.utc).isoformat()
refs={r:git('rev-parse',r).decode().strip() for r in ['origin/agent-1','origin/agent-2','origin/agent-3','origin/main']}
assert refs['origin/agent-1']=='1dd897c4467099e7cb4d7494e2958a991137df84'
assert refs['origin/agent-2']=='09dbb5be7811d7951843cef57ce69198f7fafe0e'
owners={}
for ref, module, base in [('origin/agent-1','ledger','4b35ad3f4d6b0f5d6b9d8d0e35a04d7a4550672d'),('origin/agent-2','protocol','376347d9b0979d1e46824a30197b935ffe91e4fb')]:
    tip=refs[ref]
    paths=git('ls-tree','-r','--name-only',tip,'--',module).decode().splitlines()
    blobs={p:git('show',f'{tip}:{p}') for p in paths}
    manifests=[f'{module}/evidence/source-manifest.sha256']
    if module=='protocol': manifests.append('protocol/f0/MANIFEST.sha256')
    checked={}
    for m in manifests:
        entries={}
        for line in blobs[m].decode().splitlines():
            h,p=line.split(None,1);p=p.lstrip('*')
            assert sha(blobs[p])==h,(m,p)
            entries[p]=h
        checked[m]={'sha256':sha(blobs[m]),'entries':len(entries)}
    delta=git('diff','--name-only',base,tip,'--',module).decode().splitlines()
    if module=='ledger':
        prod=[p for p in paths if p.startswith(('ledger/src/','ledger/script/')) or p=='ledger/foundry.toml']
        assert all(blobs[p]==git('show',f'933d2a7:{p}') for p in prod)
    else:
        prod=[p for p in paths if not p.startswith('protocol/f0/')]
        assert not git('diff',base,tip,'--',module,':(exclude)protocol/f0').strip()
    owners[module]={'commit':tip,'files':len(paths),'manifests':checked,'deltaSince':base,'delta':delta,'unchangedComparedFiles':len(prod),'blobs':{p:sha(b) for p,b in blobs.items()}}
save('owner-review.json',{'capturedAt':now(),'refs':refs,'owners':owners,'commits':{r:git('show','-s','--format=fuller',t).decode() for r,t in refs.items()},'classification':'exact source and published evidence audit; not fresh live execution'})
# File-changing commit provenance and partial-order conflict detection.
paths=sorted(set(p for t in refs.values() for p in git('ls-tree','-r','--name-only',t,'--','communication').decode().splitlines()))
records={};conflicts=[];ancestry={}
def ancestor(x,y):
    if (x,y) not in ancestry:
        ancestry[x,y]=subprocess.run(['git','merge-base','--is-ancestor',x,y],cwd=R).returncode==0
    return ancestry[x,y]
for p in paths:
    cs=[]
    for ref,t in refs.items():
        if subprocess.run(['git','cat-file','-e',f'{t}:{p}'],cwd=R,stdout=subprocess.DEVNULL,stderr=subprocess.DEVNULL).returncode: continue
        c=git('log','-1','--format=%H',t,'--',p).decode().strip()
        cs.append({'ref':ref,'tip':t,'changeCommit':c,'sha256':sha(git('show',f'{t}:{p}'))})
    maximal=[x for x in cs if not any(x['changeCommit']!=y['changeCommit'] and ancestor(x['changeCommit'],y['changeCommit']) for y in cs)]
    if len(set(x['sha256'] for x in maximal))>1:conflicts.append(p)
    records[p]={'candidates':cs,'latest':maximal}
save('communication-provenance.json',{'capturedAt':now(),'refs':refs,'files':records,'incomparableContentConflicts':conflicts})
assert not conflicts,conflicts

original=json.loads((O.parent/'owner-review.json').read_text())
model=[p for p in owners['protocol']['blobs'] if p.startswith(('protocol/f0/model/','protocol/f0/evidence/'))]
assert all(owners['protocol']['blobs'][p]==original['owners']['protocol']['blobs'][p] for p in model)
save('revision-check.json',{'capturedAt':now(),'prior':'39af20f58b6f2838e740697e3dc6ca86abbe87cd','current':refs['origin/agent-2'],'unchangedModelAndEvidenceFiles':model,'changedProtocolPaths':git('diff','--name-only','39af20f58b6f2838e740697e3dc6ca86abbe87cd',refs['origin/agent-2'],'--','protocol').decode().splitlines(),'modelRerunRepeated':False})
print(json.dumps({'revision2':'AUDITED','protocolFiles':owners['protocol']['files'],'communicationPaths':len(records),'modelEvidenceUnchanged':len(model),'conflicts':conflicts}))

ackpath='communication/agent-1/outbox/0095-ack-a2-f0-revision2-ledger-scope.md'
ackblob=git('show',refs['origin/agent-1']+':'+ackpath)
expected={p:owners['protocol']['blobs'][p] for p in ['protocol/f0/MANIFEST.sha256','protocol/f0/SIGNING-RULES.md','protocol/f0/D50-DEPLOYMENT-BINDING.md','protocol/f0/INVENTORY.md']}
assert all(h.encode() in ackblob for h in expected.values())
save('scoped-ack.json',{'messageId':'A1-0095','commit':refs['origin/agent-1'],'path':ackpath,'sha256':sha(ackblob),'reviewedProtocolCommit':refs['origin/agent-2'],'reviewedFiles':expected,'scope':'ledger-facing review only; excludes protocol internals/model/layout and all F0/scope/migration/interface/execution freeze ACKs','executionAuthorized':False,'agent3Ack':None})
