import datetime, hashlib, json, pathlib, subprocess, sys, time

R = pathlib.Path('/home/karma/Documents/RIS')
O = R / 'evaluation/evidence/reviews/f0-protocol-input-20261103'
O.mkdir(parents=True, exist_ok=True)
def git(*args):
    return subprocess.check_output(['git', *args], cwd=R)
def sha(b): return hashlib.sha256(b).hexdigest()
def save(name, obj): (O/name).write_text(json.dumps(obj, indent=2)+'\n')
def now(): return datetime.datetime.now(datetime.timezone.utc).isoformat()
refs={r:git('rev-parse',r).decode().strip() for r in ['origin/agent-1','origin/agent-2','origin/agent-3','origin/main']}
assert refs['origin/agent-1']=='2ba85aa2ab3ea476f972036e7e89de5bcee38497'
assert refs['origin/agent-2']=='39af20f58b6f2838e740697e3dc6ca86abbe87cd'
owners={}
for ref, module, base in [('origin/agent-1','ledger','4b35ad3f4d6b0f5d6b9d8d0e35a04d7a4550672d'),('origin/agent-2','protocol','376347d9b0979d1e46824a30197b935ffe91e4fb')]:
    tip=refs[ref]
    paths=git('ls-tree','-r','--name-only',tip,'--',module).decode().splitlines()
    blobs={p:git('show',f'{tip}:{p}') for p in paths}
    manifests=[f'{module}/evidence/source-manifest.sha256']
    if module=='protocol': manifests.append('protocol/f0/MANIFEST.sha256')
    checked={}
    for m in manifests:
        entries={}
        for line in blobs[m].decode().splitlines():
            h,p=line.split(None,1);p=p.lstrip('*')
            assert sha(blobs[p])==h,(m,p)
            entries[p]=h
        checked[m]={'sha256':sha(blobs[m]),'entries':len(entries)}
    delta=git('diff','--name-only',base,tip,'--',module).decode().splitlines()
    if module=='ledger':
        prod=[p for p in paths if p.startswith(('ledger/src/','ledger/script/')) or p=='ledger/foundry.toml']
        assert all(blobs[p]==git('show',f'933d2a7:{p}') for p in prod)
    else:
        prod=[p for p in paths if not p.startswith('protocol/f0/')]
        assert not git('diff',base,tip,'--',module,':(exclude)protocol/f0').strip()
    owners[module]={'commit':tip,'files':len(paths),'manifests':checked,'deltaSince':base,'delta':delta,'unchangedComparedFiles':len(prod),'blobs':{p:sha(b) for p,b in blobs.items()}}
save('owner-review.json',{'capturedAt':now(),'refs':refs,'owners':owners,'commits':{r:git('show','-s','--format=fuller',t).decode() for r,t in refs.items()},'classification':'exact source and published evidence audit; not fresh live execution'})
# File-changing commit provenance and partial-order conflict detection.
paths=sorted(set(p for t in refs.values() for p in git('ls-tree','-r','--name-only',t,'--','communication').decode().splitlines()))
records={};conflicts=[];ancestry={}
def ancestor(x,y):
    if (x,y) not in ancestry:
        ancestry[x,y]=subprocess.run(['git','merge-base','--is-ancestor',x,y],cwd=R).returncode==0
    return ancestry[x,y]
for p in paths:
    cs=[]
    for ref,t in refs.items():
        if subprocess.run(['git','cat-file','-e',f'{t}:{p}'],cwd=R,stdout=subprocess.DEVNULL,stderr=subprocess.DEVNULL).returncode: continue
        c=git('log','-1','--format=%H',t,'--',p).decode().strip()
        cs.append({'ref':ref,'tip':t,'changeCommit':c,'sha256':sha(git('show',f'{t}:{p}'))})
    maximal=[x for x in cs if not any(x['changeCommit']!=y['changeCommit'] and ancestor(x['changeCommit'],y['changeCommit']) for y in cs)]
    if len(set(x['sha256'] for x in maximal))>1:conflicts.append(p)
    records[p]={'candidates':cs,'latest':maximal}
save('communication-provenance.json',{'capturedAt':now(),'refs':refs,'files':records,'incomparableContentConflicts':conflicts})
assert not conflicts,conflicts
# Re-run the owner's corrected bounded design check in a pinned scratch snapshot.
scratch=R/'.git/ris-f0-protocol-input-snapshot'
scratch.mkdir(exist_ok=True)
modelpaths=[p for p in owners['protocol']['blobs'] if p.startswith('protocol/f0/')]
for p in modelpaths:
    dest=scratch/p;dest.parent.mkdir(parents=True,exist_ok=True)
    dest.write_bytes(git('show',f"{refs['origin/agent-2']}:{p}"))
before={p:sha((scratch/p).read_bytes()) for p in modelpaths}
argv=[sys.executable,'-I','conflict_key_model.py'];started=now();clock=time.monotonic()
run=subprocess.run(argv,cwd=scratch/'protocol/f0/model',stdout=subprocess.PIPE,stderr=subprocess.PIPE)
(O/'model-rerun.stdout.json').write_bytes(run.stdout)
(O/'model-rerun.stderr.txt').write_bytes(run.stderr)
assert run.returncode==0,run.stderr.decode()
observed=json.loads(run.stdout)
expected=json.loads((scratch/'protocol/f0/evidence/model-run2-corrected.json').read_text())
assert observed==expected
assert before=={p:sha((scratch/p).read_bytes()) for p in modelpaths}
save('model-rerun.json',{'startedAt':started,'endedAt':now(),'wallSeconds':time.monotonic()-clock,'argv':argv,'cwd':str(scratch/'protocol/f0/model'),'exitCode':run.returncode,'python':sys.version,'stdoutSha256':sha(run.stdout),'stderrSha256':sha(run.stderr),'ownerOutputSemanticEqual':True,'snapshotFilesStable':before,'scope':'owner design abstraction rerun only; no TLC, validation, RPC, implementation or gate claim'})
# Check published masked-code outputs arithmetically; the deployments are unavailable.
reports={}
for p in owners['ledger']['blobs']:
    if p.startswith('ledger/evidence/f0-d50-masked/') and p.endswith('.json'):
        b=git('show',f"{refs['origin/agent-1']}:{p}");d=json.loads(b)
        assert d['summary']['allMaskedMatch']
        for result in d['results']:
            assert result['allMatch']
            for row in result['contracts']:
                assert row['maskedEqualsCompiled'] and row['maskedLiveSha256']==row['compiledSha256']==row['manifestRuntimeSha256']
        reports[p]={'sha256':sha(b),'summary':d['summary'],'contracts':sum(len(x['contracts']) for x in d['results'])}
assert len(reports)==4
save('masked-report-audit.json',{'capturedAt':now(),'reports':reports,'scope':'internal consistency of owner reports only; no raw bytecode/Forge artifacts or deployment files supplied; live checks NOT rerun; Governance unmeasured'})
print(json.dumps({'manifests':{k:v['manifests'] for k,v in owners.items()},'communicationPaths':len(records),'conflicts':conflicts,'modelOutputEqual':True,'maskedReportsConsistent':len(reports)}))
