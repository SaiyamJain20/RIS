import gzip,hashlib,json,os,re,subprocess,sys
from pathlib import Path
ROOT=Path('/home/karma/Documents/RIS');OUT=ROOT/'evaluation/evidence/reviews/a2-runtime-review-20261113';RUN=ROOT/'evaluation/runs/a2-runtime-review-20261113'
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def read(n):return json.loads((OUT/n).read_text())
before=read('inputs-before.json');owner=read('owner-source.json');rev=read('helper-revisions.json')
assert subprocess.check_output(['git','rev-parse','HEAD'],cwd=ROOT).decode().strip()==before['head']

for p,h in owner['paths'].items():assert sha(RUN/'owner'/p)==h,p

for p,h in rev['after'].items():assert sha(ROOT/p)==h,p

assert '2 failed, 2935 passed, 201 skipped, 267 deselected' in log('isolated-suite-corrected')
