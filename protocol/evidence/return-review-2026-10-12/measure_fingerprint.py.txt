"""Deploy several ledger profiles on FRESH Anvil pairs (same recipe as protocol/tests/conftest.py) and print each deployment's worker fingerprint."""
import hashlib, json, os, pathlib, socket, subprocess, sys, tarfile, tempfile, time, urllib.request
REPO = pathlib.Path(sys.argv[1]); COMMIT = "91add8cc6000d1413bad2ed0aa119a5f26e8c776"
FOUNDRY = pathlib.Path(os.environ["HMBC_FOUNDRY_DIR"]); KEY0 = "0xac0974bec39a17e36ba4a6b4d238ff944bacb478cbed5efcae784d7bf4f2ff80"
def port():
    with socket.socket() as s: s.bind(("127.0.0.1", 0)); return s.getsockname()[1]
def rpc(u):
    b = json.dumps({"jsonrpc":"2.0","id":1,"method":"eth_chainId","params":[]}).encode()
    return json.loads(urllib.request.urlopen(urllib.request.Request(u, data=b, headers={"content-type":"application/json"}), timeout=5).read())
def fp(c):
    return hashlib.sha256(json.dumps({"pbc":[c["pbc"]["chainId"], c["pbc"]["contracts"]["PSC"], c["pbc"]["contracts"]["RSC"]],
        "dbc":[c["dbc"]["chainId"], c["dbc"]["contracts"]["KSSC"], c["dbc"]["contracts"]["ACSC"]]}, sort_keys=True).encode()).hexdigest()
for profile in sys.argv[2:]:
    tmp = pathlib.Path(tempfile.mkdtemp(prefix="fp-", dir=os.environ["TMPDIR"]))
    tar = subprocess.run(["git","archive",COMMIT], cwd=REPO, capture_output=True, check=True).stdout
    (tmp/"s.tar").write_bytes(tar)
    with tarfile.open(tmp/"s.tar") as t: t.extractall(tmp/"src", filter="data")
    ports = (port(), port())
    procs = [subprocess.Popen([str(FOUNDRY/"anvil"),"--port",str(p),"--chain-id",str(c),"--silent"], stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL) for p,c in zip(ports,(18545,28545))]
    try:
        urls = [f"http://127.0.0.1:{p}" for p in ports]
        for u in urls:
            for _ in range(100):
                try: rpc(u); break
                except Exception: time.sleep(0.2)
        env = {**os.environ, "PATH": f"{FOUNDRY}:{os.environ['PATH']}", "DEPLOYER_PK": KEY0}
        r = subprocess.run(["bash","ledger/tools/deploy_local.sh",urls[0],urls[1],profile,"complete-details","fpmeasure"], cwd=tmp/"src", env=env, capture_output=True, text=True, timeout=600)
        if r.returncode != 0: print(json.dumps({"profile":profile,"error":r.stderr[-300:]})); continue
        d = json.loads((tmp/"src/ledger/exports/deployments/fpmeasure/deployment.json").read_text()); c = d["chains"]
        print(json.dumps({"profile":d["profile"],"mutations":d["mutations"],"deployBlock":[c["pbc"]["deployBlock"],c["dbc"]["deployBlock"]],
                          "PSC":c["pbc"]["contracts"]["PSC"],"KSSC":c["dbc"]["contracts"]["KSSC"],"fingerprint":fp(c)[:16]}), flush=True)
    finally:
        for p in procs: p.terminate()
        for p in procs:
            try: p.wait(timeout=10)
            except subprocess.TimeoutExpired: p.kill()
        subprocess.run(["rm","-rf",str(tmp)])
