def stage(name):
    def deco(fn):
        t0 = time.monotonic()
        try:
            result = fn()
            row = {"stage": name, "ok": True, "seconds": round(time.monotonic() - t0, 2), "result": result}
        except BaseException as e:  # noqa: BLE001 - a measurement: record whatever happened
            row = {"stage": name, "ok": False, "seconds": round(time.monotonic() - t0, 2), "error": type(e).__name__ + ": " + str(e)[:700], "traceback": traceback.format_exc()[-2000:]}
        RESULTS.append(row)
        (OUT / "results.json").write_text(json.dumps(RESULTS, indent=1, default=str) + "\n")
        print(("PASS " if row["ok"] else "FAIL ") + name + ("" if row["ok"] else "  -> " + row["error"][:240]), flush=True)
        return fn
    return deco


class Factory:
    def __init__(self, root):
        self.root, self.n = root, 0

    def mktemp(self, name):
        self.n += 1
        p = self.root / f"{name}-{self.n}"
        p.mkdir(parents=True)
        return p

def brief_reply(r):
    if r.get("ok"):
        res = r["result"]
        coll = res.get("collection", {})
        return {"ok": True, "outcomes": [(o["witnessIndex"], o["status"], o.get("replies")) for o in coll.get("outcomes", [])], "keys": sorted(res)}
    err = r["error"]
    out = {"ok": False, "code": err.get("code"), "reason": err.get("reason"), "message": str(err.get("message"))[:200]}
    if "collection" in err:
        out["outcomes"] = [(o["witnessIndex"], o["status"], o.get("code")) for o in err["collection"]["outcomes"]]
    return out

@stage("S3 decision D1 through the adapter, consumed by the real ACSC; receipt through the adapter, reported to the real PSC")
def _():
    live, led, a = S["live"], S["led"], S["adapter"]
    signed1 = S["signed"][0]
    d1 = signed1["requestDigest"]
    r = a.ask("collect_decision", {"requestDigest": d1})

    assert res["decision"]["result"] == "1" and res["credentialRecords"]
    cert = {"decision": res["decision"], "signatures": res["signatures"]}
    led.dbc.send(setup.key("relay"), led.ACSC, "consumeDecision", signed1["request"], cert, res["credentialRecords"])
    grant_id = "0x" + eip712.grant_id(bytes.fromhex(d1[2:]), led.KSSC.address, led.dbc.chain_id).hex()

    S["receipt1"] = rres["receipt"]
    return {"decisionReply": brief_reply(r), "decision": res["decision"], "receiptReply": brief_reply(r2), "receipt": rres["receipt"], "grantId": grant_id}

@stage("S9 checkpoint_rpc_v2.collect on the real checkpoint digests (sequences 1 and 2), two caller head selections: (A) the PBC latest block now, (B) each checkpoint's own source block")
def _():
    live = S["live"]
    pbc_id = live.dep["chains"]["pbc"]["chainId"]
    out = []
    for i in range(4):
        x = S["exports"][i]
        ev = x["ev"]
        rows = [(r, s) for coll in ("superseded", "conflicts") for r in ev[coll] if r["type"] == "StateCheckpoint" for s in r["digests"]]
        if not rows:
            out.append({"witness": i, "verdict": "NOT PRODUCED", "reason": "this witness's store holds no StateCheckpoint record"})
            continue
        latest = head(live.urls[0], pbc_id)
        for mode in ("A-latest-now", "B-own-source-block"):
            selections = []
            for r, s in rows:
                if mode.startswith("A"):
                    h = latest
                else:
                    blk = raw_rpc(live.urls[0], "eth_getBlockByHash", [r["effect"]["sourceBlockHash"], False])
                    h = {"chainId": pbc_id, "blockNumber": str(int(blk["number"], 16)), "blockHash": blk["hash"], "timestamp": str(int(blk["timestamp"], 16))}
                selections.append({"digest": s["digest"], "pbcHead": h, "reason": "agent 2 cross-run 2, mode " + mode + ": a caller assertion"})
            rp = {c: rpc_client(live.urls[k]) for k, c in enumerate(("pbc", "dbc"))}
            try:
                res = cr.collect(ev, manifest=live.dep, observations=S["obs"], expected_owner=x["owner"], abi_paths=S["abi"], selections=selections, rpcs=rp, output=RAW / f"checkpoint-w{i}-{mode}", compiled_max_admins="8")
                out.append({"witness": i, "mode": mode, "verdict": "ACCEPTED", "digests": len(rows), "keys": sorted(res)})
            except Exception as ex:  # noqa: BLE001
                out.append({"witness": i, "mode": mode, "verdict": "REFUSED", "digests": len(rows), "error": type(ex).__name__ + ": " + str(ex)[:500]})
    return out

def build_receipt_captures(ev, dbc_obs):
    """MY capture builder (NOT agent 3 code): agent 3 has no receipt RPC collector yet, so the closed capture records receipt_history_v2 asks for are built here from raw RPC replies."""
    live = S["live"]
    dep = live.dep
    pbc_id, dbc_id = dep["chains"]["pbc"]["chainId"], dep["chains"]["dbc"]["chainId"]
    url = {pbc_id: live.urls[0], dbc_id: live.urls[1]}
    kssc, psc = contract("KSSC"), contract("PSC")
    caps = []
    for coll in ("superseded", "conflicts"):
        for row in ev[coll]:
            if row["type"] != "GrantReceipt":
                continue
            message = row["effect"] | {"requestDigest": row["keyField"]}
            for signed in row["digests"]:
                pbc_ob = signed["observation"]
                blocks = {}

                def blk(chain, h):
                    if (chain, h) not in blocks:
                        blocks[(chain, h)] = {"chainId": chain, "raw": raw_rpc(url[chain], "eth_getBlockByHash", [h, False])}
                blk(pbc_id, pbc_ob["blockHash"])
                blk(dbc_id, dbc_obs["blockHash"])
                blk(dbc_id, message["dbcBlockHash"])
                calls = []
                for chain, ob, c, getter, arg in ((dbc_id, dbc_obs, kssc, "grantDetails", message["grantId"]), (pbc_id, pbc_ob, psc, "getRequest", message["requestDigest"])):
                    data = c.calldata(getter, arg)
                    res = raw_rpc(url[chain], "eth_call", [{"to": c.address, "data": data}, {"blockHash": ob["blockHash"]}])
                    calls.append({"chainId": chain, "blockHash": ob["blockHash"], "to": c.address, "data": data, "result": res})
                logs = [lg for lg in raw_rpc(url[dbc_id], "eth_getLogs", [{"blockHash": message["dbcBlockHash"], "address": kssc.address}])
                        if (kssc.decode_log(lg) or {}).get("name") == "GrantIssued" and kssc.decode_log(lg)["fields"]["grantId"] == message["grantId"]]
                caps.append({"digest": signed["digest"], "dbcObservation": dbc_obs, "blocks": list(blocks.values()), "calls": calls, "logs": logs})
    return caps

@stage("S13 overtaken-checkpoint attempt on REAL witness processes: a proposal for a sequence the ledger has already accepted")
def _():
    live, led, wview = S["live"], S["led"], S["wview"]
    (WORK / "raw").mkdir(exist_ok=True)
    procs = [L.WitnessProc(live, WORK / "raw", i) for i in range(3)]
    S["rawprocs"] = procs
    try:
        L.mine(live, "pbc")
        L.mine(live, "dbc")
        cp = L.svc.derive_checkpoint(wview, live.dep)
        replies = [p.ask("checkpoint", cp) for p in procs]
        assert all(r["type"] == "signature" for r in replies), replies
        ordered = L.sorted_signatures(cp, "StateCheckpoint", [r["signature"] for r in replies], live.dep)
        led.dbc.send(L.ANVIL_KEY0, led.KSSC, "acceptState", cp, ordered)
        L.mine(live, "pbc")
        L.mine(live, "dbc")
        nxt = L.svc.derive_checkpoint(wview, live.dep)                     # sequence = current + 1 (a fresh, valid proposal)
        stale_diff = dict(nxt)
        stale_diff["sequence"] = cp["sequence"]                            # the SAME sequence the ledger has accepted, a DIFFERENT effect (later source block)
        r_diff = procs[0].ask("checkpoint", stale_diff)
        r_same = procs[0].ask("checkpoint", cp)                            # the exact accepted proposal again
        return {"acceptedSequence": cp["sequence"], "staleDifferentEffect": {k: v for k, v in r_diff.items() if k in ("type", "code", "reason", "message")},
                "sameProposalAgain": {k: v for k, v in r_same.items() if k in ("type", "code", "reason", "message")}}
    finally:
        for p in procs:
            try:
                p.close()
            except Exception:  # noqa: BLE001
                pass
