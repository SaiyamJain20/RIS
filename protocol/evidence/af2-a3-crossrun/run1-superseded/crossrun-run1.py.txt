  python -I crossrun.py SCR REPO OUTDIR            (HOME and HMBC_FOUNDRY_DIR set by the caller; run under `flock -x $SCR/live.lock`)

Roles, as in a real run: two real Anvil nodes with agent 1's real F ledger; four REAL `witness_main` processes (agent 2's) kept ALIVE; agent 3's REAL `store_snapshot_v2.locked_store`,
`export_snapshot_v2.verify_export_snapshot`, `public_observations_v2.collect`, `history_rpc_v2.collect`, `checkpoint_rpc_v2.collect` (unmodified, imported from an extracted copy of origin/agent-3).

def stage(name):
    def deco(fn):
        t0 = time.monotonic()
        try:
            result = fn()
            row = {"stage": name, "ok": True, "seconds": round(time.monotonic() - t0, 2), "result": result}
        except BaseException as e:  # noqa: BLE001 - a measurement: record whatever happened
            row = {"stage": name, "ok": False, "seconds": round(time.monotonic() - t0, 2), "error": type(e).__name__ + ": " + str(e)[:600], "traceback": traceback.format_exc()[-1800:]}
        RESULTS.append(row)
        (OUT / "results.json").write_text(json.dumps(RESULTS, indent=1, default=str) + "\n")
        print(("PASS " if row["ok"] else "FAIL ") + name + ("" if row["ok"] else "  -> " + row["error"][:200]), flush=True)
        return fn
    return deco


class Factory:
    def __init__(self, root):
        self.root, self.n = root, 0

    def mktemp(self, name):
        self.n += 1
        p = self.root / f"{name}-{self.n}"
        p.mkdir(parents=True)
        return p


def sh(raw):
    return hashlib.sha256(raw).hexdigest()

def rpc_client(url):

class Pinned:
    """A transport shim, NOT agent 3 code: it answers `eth_getBlockByNumber latest` with the block of the export's own anchor. Used only for the control that shows what the anchor mismatch needs."""
    def __init__(self, inner, number):
        self.inner, self.n, self.url = inner, number, inner.url

    def call(self, method, params=None):
        params = list(params or [])
        if method == "eth_getBlockByNumber" and params and params[0] == "latest":
            params = [hex(self.n), False]
        return self.inner.call(method, params)

    def checkpoint(self, label):
        return self.inner.checkpoint(label)


def store_paths(d):
    return d / "signing.sqlite", d / "INITIALIZED", d / ".lock"


def snap(d):
    db, marker, lock = store_paths(d)
    with ss.locked_store(db, marker, lock) as r:
        s = r.snapshot()
    return s


def brief(s):
    return {"markerBytes": len(s.marker), "markerSha256": s.marker_sha256[:16], "logicalSha256": s.logical_sha256[:16], "walPresent": s.wal_sha256 is not None}


def run_exporter(index, flag):

                       cwd=PROTO, env={"PATH": "/usr/bin:/bin"}, capture_output=True, timeout=60)
    return p.stdout, p.stderr, p.returncode

@stage("S1 start four real witness_main processes (kept alive) and wait for INITIALIZED")
def _():
    live = S["live"]
    S["tmp"] = tmp = WORK / "w"
    tmp.mkdir()
    S["procs"] = [L.WitnessProc(live, tmp, i) for i in range(4)]
    S["dirs"] = [tmp / "wstate" / f"witness-{i}" for i in range(4)]
    deadline = time.monotonic() + 90
    while not all((d / "INITIALIZED").exists() for d in S["dirs"]):
        if time.monotonic() > deadline:
            raise RuntimeError("stores not initialised in 90 s")
        for p in S["procs"]:
            if p.p.poll() is not None:
                raise RuntimeError("a witness died: " + p.p.stderr.read()[-500:])
        time.sleep(0.3)
    return {"files": sorted(os.listdir(S["dirs"][0])), "markerBytes": (S["dirs"][0] / "INITIALIZED").read_bytes().decode()[:200]}

@stage("S5 agent 3 public_observations_v2.collect against the live ledger (latest anchor), then the exporter choreography under locked_store for each witness")
def _():
    res, rp = collect_latest()
    S["obs_latest"] = res["observations"]
    S["rp"] = rp
    live = S["live"]
    out = []
    for i, d in enumerate(S["dirs"]):
        db, marker, lock = store_paths(d)
        owner = "witness:" + live.dep["fixture"]["witnessSet"]["witnesses"][i]
        with ss.locked_store(db, marker, lock) as r:
            before = r.snapshot()
            b_out, b_err, b_rc = run_exporter(i, "--binding")
            e_out, e_err, e_rc = run_exporter(i, "--evidence")
            after = r.snapshot()
            ss.require_unchanged(before, after)
        (RAW / f"w{i}-binding.stdout").write_bytes(b_out)
        (RAW / f"w{i}-evidence.stdout").write_bytes(e_out)
        S.setdefault("exports", {})[i] = dict(binding=(b_out, b_err, b_rc), evidence=(e_out, e_err, e_rc), snapshot=before, owner=owner)
        out.append({"witness": i, "bindingExit": b_rc, "bindingStderrBytes": len(b_err), "bindingStdoutBytes": len(b_out), "evidenceExit": e_rc, "evidenceStderrBytes": len(e_err), "evidenceStdoutBytes": len(e_out)})
    return {"observationAnchors": {c: S["obs_latest"][c]["anchor"] for c in ("pbc", "dbc")}, "exports": out}

@stage("S9 a signing request that arrives while agent 3 holds the exclusive snapshot lock: does the live witness wait (cooperative), refuse, or sign?")
def _():
    live = S["live"]
    wview, _chains = L.witness_view(live)
    L.mine(live, "dbc")
    cp2 = L.svc.derive_checkpoint(wview, live.dep)
    d = S["dirs"][3]
    db, marker, lock = store_paths(d)
    hold = 4.0
    state = {}

    def holder():
        with ss.locked_store(db, marker, lock):
            state["locked"] = time.monotonic()
            time.sleep(hold)
        state["released"] = time.monotonic()
    t = threading.Thread(target=holder)
    t.start()
    while "locked" not in state:
        time.sleep(0.02)
    t0 = time.monotonic()
    reply = S["procs"][3].ask("checkpoint", cp2)
    t1 = time.monotonic()
    t.join()
    return {"holdSeconds": hold, "replyType": reply["type"], "askReturnedAfterSeconds": round(t1 - t0, 2), "askReturnedAfterRelease": t1 >= state["released"] - 0.05}

    return subprocess.Popen([sys.executable, "-B", "-c", PROBE, str(PROTO), str(w), str(dep_file), mode], stdin=subprocess.PIPE, stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True)


def line_within(p, seconds):
    box = {}

    def rd():
        box["line"] = p.stdout.readline().strip()
    t = threading.Thread(target=rd, daemon=True)
    t.start()
    t.join(seconds)
    return box.get("line")


@stage("S11 worker lock honouring: with agent 3 holding `.init.lock` exclusively, (a) a Worker being constructed, (b) an already running Worker's record write")
def _():
    w = S["worker_dir"]
    db, marker, lock = w / "worker.sqlite", w / "INITIALIZED", w / ".init.lock"
    res = {}
    # (b) a worker constructed BEFORE the snapshot lock is taken (how a running adapter looks), then writes while the lock is held
    pre = probe(w, "write-on-go")
    res["preConstructed"] = line_within(pre, 20)
    outcome = {}
    with ss.locked_store(db, marker, lock) as r:
        before = r.snapshot()
        pre.stdin.write("go\n")
        pre.stdin.flush()
        outcome["writeWhileLocked"] = line_within(pre, 5)
        after = r.snapshot()
        try:
            ss.require_unchanged(before, after)
            outcome["beforeAfterCheck"] = "UNCHANGED (the write was not visible or did not happen)"
        except ValueError as ex:
            outcome["beforeAfterCheck"] = "agent 3 detects it: " + str(ex)
    res["runningWorker"] = outcome
    pre.stdin.close()
    pre.wait(timeout=20)
    # (a) a Worker constructed while the lock is held
    state = {}

    def holder():
        with ss.locked_store(db, marker, lock):
            state["held"] = time.monotonic()
            time.sleep(3.0)
        state["released"] = time.monotonic()
    t = threading.Thread(target=holder)
    t.start()
    while "held" not in state:
        time.sleep(0.02)
    p = probe(w, "construct-only")
    t0 = time.monotonic()
    line = line_within(p, 20)
    res["constructWhileLocked"] = {"line": line, "returnedAfterSeconds": round(time.monotonic() - t0, 2), "lockHeldSeconds": 3.0, "waitedForRelease": time.monotonic() - state["held"] >= 2.9}
    t.join()
    p.stdin.close()
    p.wait(timeout=20)
    return res
