"""Protocol-scope review checks of the SL7 public evidence (S3-0020). Read-only; prints one JSON report.
Usage: python -I review_sl7.py <evidence dir> <repo>/protocol"""
import json
import pathlib
import re
import sys

sys.path.insert(0, sys.argv[2])
from eth_utils import keccak  # noqa: E402
from hmbc_protocol import eip712, keys, messages  # noqa: E402
from hmbc_protocol.abi import Contract  # noqa: E402

EVD = pathlib.Path(sys.argv[1]); RUN = EVD / "run"; PROTO = pathlib.Path(sys.argv[2])
EMPTY = "e3b0c44298fc1c149afbf4c8996fb92427ae41e4649b934ca495991b7852b855"
NEVER = re.compile(r"no worker\.sqlite in (\S+) \(looked in <dir>/worker/ and in <dir>\)\n")
load = lambda p: json.loads(pathlib.Path(p).read_text())  # noqa: E731
PSC = Contract(json.loads((PROTO / "abi/PSC.json").read_text()))
report, problems = {}, []


def check(c, m):
    if not c:
        problems.append(m)


def walk(o):
    yield o
    if isinstance(o, dict):
        for v in o.values():
            yield from walk(v)
    elif isinstance(o, list):
        for v in o:
            yield from walk(v)


def signer(name, msg, sig, dep):
    return eip712.recover(messages.digest_of(name, msg, messages.domain_for(name, msg, dep)), sig)


def adapter_ok(p, case, name):
    check(p["argv"][1:3] == ["-m", "hmbc_protocol.adapter"] and p["exitCode"] == 0 and p["failure"] is None
          and p.get("stderrSha256") == EMPTY and not [e for e in p["exchanges"] if e["failure"]], f"{case} {name} adapter")


def request_check(case, txfile, prepared, dep, label):
    txs = load(txfile)
    ra = [t for t in txs if t.get("method") == "requestAccess"]
    check(len(ra) == 1 and ra[0]["signerRole"] == "relay", f"{case} one relay requestAccess on {label}")
    _, args = PSC.decode_input(bytes.fromhex(ra[0]["calldata"][2:]))
    on, sig = args[0], args[1]
    dig = "0x" + messages.digest_of("AccessRequest", on, messages.domain_for("AccessRequest", on, dep)).hex()
    check(on == prepared["request"] and sig == prepared["signature"] and dig == prepared["requestDigest"]
          and eip712.recover(bytes.fromhex(dig[2:]), sig) == on["gid"], f"{case} {label} calldata = prepared")
    c = dep["chains"]
    check(on["psc"] == c["pbc"]["contracts"]["PSC"] and on["kssc"] == c["dbc"]["contracts"]["KSSC"]
          and on["sourceChainId"] == c["pbc"]["chainId"] and on["destinationChainId"] == c["dbc"]["chainId"], f"{case} {label} binding")
    return on, dig


probes_total = 0
for case in sorted(p.name for p in RUN.iterdir() if p.is_dir()):
    profile = case.split("-wrong")[0].split("-expiry")[0]
    wrong, repaired = "wrong-destination" in case, profile == "repaired-e"
    row, r = RUN / case / profile, {}
    d1 = load(row / "adapter/process.json"); a = load(row / "destination-instance/adapter/process.json")
    adapter_ok(d1, case, "d1"); adapter_ok(a, case, "A")
    sd = lambda p: p["argv"][p["argv"].index("--state-dir") + 1]  # noqa: E731
    dp = lambda p: p["argv"][p["argv"].index("--deployment") + 1]  # noqa: E731
    dep_a, dep_1 = load(row / "destination-instance/deployment.json"), load(row / "deployment.json")
    states = [sd(d1), sd(a)]
    exa_f = row / "destination-stress/exchanges.json"
    ex_a = load(exa_f) if exa_f.exists() else []
    check(len(ex_a) == len(a["exchanges"]), f"{case} A exchange count")
    decisions = [e for e in ex_a if e["operation"] == "collect_decision"]
    r["A_ops"] = [e["operation"] for e in ex_a]
    if wrong:
        b = load(row / "source-instance/adapter/process.json"); adapter_ok(b, case, "B")
        dep_b = load(row / "source-instance/deployment.json")
        states.append(sd(b))
        check(len(set(states)) == 3 and dp(b) != dp(a), f"{case} three distinct states; B pinned to its own deployment")
        check(all(dep_b["chains"][n]["chainId"] == dep_a["chains"][n]["chainId"] for n in ("pbc", "dbc"))
              and all(not set(dep_b["chains"][n]["contracts"].values()) & (set(dep_a["chains"][n]["contracts"].values()) | set(dep_1["chains"][n]["contracts"].values()))
                      for n in ("pbc", "dbc")), f"{case} B: same chain pair, distinct contracts")
        ex_b = load(row / "destination-stress/source/exchanges.json")
        check(len(ex_b) == len(b["exchanges"]) and all(e["response"]["ok"] for e in ex_b), f"{case} B exchanges")
        r["B_ops"] = [e["operation"] for e in ex_b]
        check(r["A_ops"] == [], f"{case} A serves no exchange in wrong-destination")
        check(r["B_ops"] == (["prepare_request", "collect_checkpoint", "collect_decision"] if repaired else ["prepare_request"]), f"{case} B ops")
        decisions += [e for e in ex_b if e["operation"] == "collect_decision"]
        on, dig = request_check(case, row / "source-instance/pbc-native/transactions.json", ex_b[0]["response"]["result"], dep_b, "B")
        a_reqs = [t for t in load(row / "destination-instance/pbc-native/transactions.json") if t.get("method") == "requestAccess"]
        check(not a_reqs, f"{case} no requestAccess on A's PBC")
        before_a = load(row / "destination-stress/before.json")
        check(before_a["requests"][dig]["status"] == "0", f"{case} A's PSC never saw the request")
        r["request"] = dict(psc=on["psc"], kssc=on["kssc"], bPsc=dep_b["chains"]["pbc"]["contracts"]["PSC"], aPsc=dep_a["chains"]["pbc"]["contracts"]["PSC"])
        if repaired:
            cpr = [e for e in ex_b if e["operation"] == "collect_checkpoint"][0]["response"]["result"]
            dr = decisions[0]["response"]["result"]
            w = dep_b["fixture"]["witnessSet"]["witnesses"][0]
            check(signer("StateCheckpoint", cpr["checkpoint"], cpr["signatures"][0], dep_b) == w
                  and signer("AccessDecision", dr["decision"], dr["signatures"][0], dep_b) == w, f"{case} B witness signatures (B domain)")
            r["bDecisionUnderADomainIsWitness"] = signer("AccessDecision", dr["decision"], dr["signatures"][0], dep_a) == w
            check(r["bDecisionUnderADomainIsWitness"] is False, f"{case} B's certificate is domain-bound to B")
    else:
        check(r["A_ops"] == (["prepare_request", "collect_checkpoint", "collect_decision"] if repaired else ["prepare_request"]), f"{case} A ops")
        check(len(set(states)) == 2, f"{case} distinct states")
        on, dig = request_check(case, row / "destination-instance/pbc-native/transactions.json", ex_a[0]["response"]["result"], dep_a, "A")
        tl, st = load(row / "destination-stress/timeline.json"), load(row / "destination-stress/stress-transaction.json")
        consume = int(st["blockTimestamp"], 16)
        check(consume == int(on["expiresAt"]) == int(tl["consumeTimestamp"]) == int(tl["requestExpiresAt"]), f"{case} consume block time == request expiry")
        r["boundary"] = dict(requestExpiresAt=on["expiresAt"], consumeBlockTimestamp=str(consume), dbcBefore=tl["dbcBefore"])
        if repaired:
            cpr = [e for e in ex_a if e["operation"] == "collect_checkpoint"][0]["response"]["result"]
            dr = decisions[0]["response"]["result"]; d, cp = dr["decision"], cpr["checkpoint"]
            w = dep_a["fixture"]["witnessSet"]["witnesses"][0]
            check(signer("StateCheckpoint", cp, cpr["signatures"][0], dep_a) == w and signer("AccessDecision", d, dr["signatures"][0], dep_a) == w, f"{case} A witness signatures")
            ttl = int(dep_a["fixture"]["ttlSeconds"]["checkpointMax"])
            witness_now_at_cp = int(cp["expiresAt"]) - ttl          # witness.py:65: expiresAt = DBC now + checkpointMax
            check(d["stateSequence"] == cp["sequence"] and d["sourceBlockNumber"] == cp["sourceBlockNumber"] and d["requestDigest"] == dig
                  and int(d["expiresAt"]) == int(on["expiresAt"]) == min(int(on["expiresAt"]), int(cp["expiresAt"]))
                  and witness_now_at_cp + 30 > int(on["expiresAt"]) and int(cp["expiresAt"]) > int(on["expiresAt"]), f"{case} decision expiry rule")
            r["boundary"].update(decisionExpiresAt=d["expiresAt"], checkpointExpiresAt=cp["expiresAt"], witnessDbcNowAtCheckpoint=str(witness_now_at_cp),
                                 lowerBoundNowPlus30=str(witness_now_at_cp + 30), timelineDecision=tl["decisionExpiresAt"], timelineCheckpoint=tl["checkpointExpiresAt"])
            check(tl["decisionExpiresAt"] == d["expiresAt"] and tl["checkpointExpiresAt"] == cp["expiresAt"], f"{case} timeline joins witness outputs")
    r["decisionCollectionsAB"] = len(decisions)
    check(len(decisions) == (1 if repaired else 0), f"{case} A31 count")
    res = load(row / "destination-stress/result.json")
    check(res["decisionCollections"] == len(decisions), f"{case} result decisionCollections")
    # D49
    probes = [c for c in load(row / "commands.json") if c.get("workerObservation")]
    probes_total += len(probes)
    check(probes and all(c["exitCode"] == 1 and c["workerObservation"] == "never-instantiated (verified)" for c in probes), f"{case} probes")
    obs = [o for f in (row / "destination-stress").rglob("*.json") for o in walk(load(f)) if isinstance(o, dict) and "instantiated" in o]
    by_dir = {}
    for o in obs:
        m = NEVER.fullmatch(o.get("message", ""))
        check(o.get("instantiated") is False and m is not None and m.group(1) in states[1:], f"{case} observation state dir")
        by_dir.setdefault(m.group(1) if m else None, []).append(o)
    check(all(all(x == v[0] for x in v) for v in by_dir.values()), f"{case} observations identical per worker")
    r["workerProbes"] = len(probes); r["observedWorkers"] = len(by_dir)
    report[case] = r
report["workerProbesTotal"] = probes_total

labels = ["witness-1", "relay", "alice", "bob", "mallory"] + [f"H{i}-{s}" for i in range(1, 5) for s in ("root", "ca-admin")]
keyhex = {l: keys.synthetic_private_key(l)[2:] for l in labels}
digs, hexes, files, tokfields, keyhits, src_files = set(), set(), 0, 0, [], 0
for f in RUN.rglob("*.json"):
    try:
        o = json.loads(f.read_text())
    except Exception:
        continue
    for x in walk(o):
        if isinstance(x, dict):
            if "tokenDigest" in x:
                digs.add(x["tokenDigest"])
            if isinstance(x.get("export"), str):
                try:
                    digs |= {y["tokenDigest"] for y in json.loads(x["export"]).get("releases", [])}
                except Exception:
                    pass
for f in EVD.rglob("*"):
    if f.is_file():
        files += 1; src_files += "source-instance" in str(f) or "/source/" in str(f)
        t = f.read_text(errors="replace").lower()
        hexes |= set(re.findall(r"0x[0-9a-f]{64}", t)); tokfields += len(re.findall(r'"token"\s*:', t))
        keyhits += [(str(f.relative_to(EVD)), l) for l, k in keyhex.items() if k in t]
hits = [x for x in hexes if "0x" + keccak(bytes.fromhex(x[2:])).hex() in digs]
check(not keyhits and not hits and tokfields == 0, "never-publish scan")
report["neverPublishScan"] = dict(files=files, sourceMaterialFiles=src_files, keyLabels=len(labels), keyHits=keyhits, hex32=len(hexes),
                                  tokenDigests=len(digs), preimageHits=len(hits), tokenFields=tokfields)
report["problems"] = problems
print(json.dumps(report, indent=1, default=str))
