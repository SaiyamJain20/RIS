def check(c, m):
    if not c:
        problems.append(m)
    return bool(c)


def walk(o):
    yield o
    if isinstance(o, dict):
        for v in o.values():
            yield from walk(v)
    elif isinstance(o, list):
        for v in o:
            yield from walk(v)


def signer(name, msg, sig, dep):
    return eip712.recover(messages.digest_of(name, msg, messages.domain_for(name, msg, dep)), sig)

def expected(sl, case, rep):
    """(stress decisions, never-instantiated probes, instantiated probes) per row, from my slice reviews S2-0005..S2-0020."""
    if sl in PRESOLO:
        return 0, 0, 0
    if sl == "sl1-subject":
        return (1 if rep else 0), 0, 0
    if sl == "sl2-relay":
        return ((2 if "same-nonce" in case else 1) if rep else 0), 2, 0
    if sl == "sl3-audit":
        return ((1 if rep else 0), 0, 2) if "dropped" in case else (0, 2, 0)
    if sl == "sl4-receipt":
        return 1, 6, 0
    if sl == "sl5-trust":
        return 0, (5 if "rollback" in case else 3), 0
    if sl == "sl6-barrier":
        return (1 if rep else 0), (6 if ("expired" in case or (rep and "post-" in case)) else 5), 0
    if sl == "sl7-destination":
        return (1 if rep else 0), (10 if "wrong-destination" in case else 5), 0
    raise ValueError(sl)
